93  template <typename T1, typename T2>
94  bool selectionTopol(const T1 &candidate, const T2 &trackPositron,
95  const T2 &trackElectron) {
96  auto candpT = candidate.pt();
97  auto pTBin = findBin(pTBins, candpT);
98  if (pTBin == -1) {
99  return false;
100  }
101 
102  // check that the candidate pT is within the analysis range
103  if (candpT < d_pTCandMin || candpT >= d_pTCandMax) {
104  return false;
105  }
106 
107  // cut on invariant mass
108  if (std::abs(InvMassJpsiToEE(candidate) -
109  RecoDecay::getMassPDG(pdg::Code::kJpsi)) >
110  cuts->get(pTBin, "m")) {
111  return false;
112  }
113 
114  // cut on daughter pT
115  if (trackElectron.pt() < cuts->get(pTBin, "pT El") ||
116  trackPositron.pt() < cuts->get(pTBin, "pT El")) {
117  return false;
118  }
119 
120  // cut on daughter DCA - need to add secondary vertex constraint here
121  if (std::abs(trackElectron.dcaPrim0()) > cuts->get(pTBin, "DCA_xy") ||
122  std::abs(trackPositron.dcaPrim0()) > cuts->get(pTBin, "DCA_xy")) {
123  return false;
124  }
125 
126  // cut on daughter DCA - need to add secondary vertex constraint here
127  if (std::abs(trackElectron.dcaPrim1()) > cuts->get(pTBin, "DCA_z") ||
128  std::abs(trackPositron.dcaPrim1()) > cuts->get(pTBin, "DCA_z")) {
129  return false;
130  }
131 
132  return true;
133  }
134  template <typename T> bool validRICHPID(const T &track) {
135  if (track.pt() < d_pidRICHMinpT || track.pt() >= d_pidRICHMaxpT) {
136  return false;
137  }
138  return true;
139  }

141  using Trks = soa::Join<aod::BigTracksPID, aod::RICHTracksIndex>;
142  void process(aod::HfCandProng2 const &candidates, const Trks &tracks,
143  aod::RICHs const &) {
144  TrackSelectorPID selectorElectron(kElectron);
145  selectorElectron.setRangePtTOF(d_pidTOFMinpT, d_pidTOFMaxpT);
146  selectorElectron.setRangeNSigmaTOF(-d_nSigmaTOF, d_nSigmaTOF);
147  selectorElectron.setRangeNSigmaTOFCondTPC(-d_nSigmaTOFCombined,
148  d_nSigmaTOFCombined);
149  // looping over 2-prong candidates
150  for (auto &candidate : candidates) {
151 
152  auto trackPos = candidate.index0_as<Trks>(); // positive daughter
153  auto trackNeg = candidate.index1_as<Trks>(); // negative daughter
154 
155  if (!(candidate.hfflag() & 1 << DecayType::JpsiToEE)) {
156  hfSelJpsiToEECandidate(0);
157  continue;
158  }
159 
160  // track selection level need to add special cuts (additional cuts on
161  // decay length and d0 norm)
162 
163  if (!selectionTopol(candidate, trackPos, trackNeg)) {
164  hfSelJpsiToEECandidate(0);
165  continue;
166  }
167 
168  // track-level PID selection
169  if (selectorElectron.getStatusTrackPIDTOF(trackPos) ==
170  TrackSelectorPID::Status::PIDRejected ||
171  selectorElectron.getStatusTrackPIDTOF(trackNeg) ==
172  TrackSelectorPID::Status::PIDRejected) {
173  hfSelJpsiToEECandidate(0);
174  continue;
175  }
176 
177  bool pidrichPos = validRICHPID(trackPos);
178  bool pidrichNeg = validRICHPID(trackNeg);
179  if (pidrichPos && pidrichNeg) {
180  // LOGF(info, "both good rich tracks");
181  }
182 
183  // LOGF(info, "cut %f, val pos %f, val neg %f", d_nSigmaRICH,
184  // trackPos.rich().richNsigmaEl(), trackNeg.rich().richNsigmaEl());
185  if (pidrichPos && pidrichNeg) {
186  bool selpidrichPos = abs(trackPos.rich().richNsigmaEl()) < d_nSigmaRICH;
187  bool selpidrichNeg = abs(trackNeg.rich().richNsigmaEl()) < d_nSigmaRICH;
188  // LOGF(info, "pos %d, neg %d", selpidrichPos, selpidrichNeg);
189  bool selpidrich = selpidrichPos && selpidrichNeg;
190  if (selpidrich) {
191  // LOGF(info, "both sel rich tracks");
192  }
193  if (!selpidrich) {
194  hfSelJpsiToEECandidate(0);
195  // LOGF(info, "not selectedtracks");
196  continue;
197  }
198  }
199 
200  hfSelJpsiToEECandidate(1);
201  }
202  }
